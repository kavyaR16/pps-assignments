#include <stdio.h>

/* Without recursion (iterative) */
long long fact_iter(int n) {
    long long f = 1;
    for (int i = 2; i <= n; i++)
        f = f * i;
    return f;
}

/* With recursion */
long long fact_rec(int n) {
    if (n == 0 || n == 1)
        return 1;
    return n * fact_rec(n - 1);
}

int main() {
    int n;
    printf("Enter a number: ");
    scanf("%d", &n);

    if (n < 0) {
        printf("Factorial is not defined for negative numbers\n");
        return 1;
    }
    printf("Factorial without recursion = %lld\n", fact_iter(n));
    printf("Factorial with recursion = %lld\n", fact_rec(n));
    return 0;
}
