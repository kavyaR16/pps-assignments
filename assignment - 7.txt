#include <stdio.h>

struct Student
{
    int roll_no;
    char name[50];
    float marks1, marks2, marks3;
    float total, percentage;
};

int main()
{
    struct Student s[100];
    int n, i;

    printf("Enter the number of students: ");
    scanf("%d", &n);

    for (i = 0; i < n; i++)
    {
        printf("\nEnter roll number, name and marks of 3 subjects:\n");
        scanf("%d %49s %f %f %f",
              &s[i].roll_no,
              s[i].name,
              &s[i].marks1,
              &s[i].marks2,
              &s[i].marks3);

        s[i].total = s[i].marks1 + s[i].marks2 + s[i].marks3;
        s[i].percentage = s[i].total / 3.0;
    }

    printf("\nRoll No\tName\tSubject1\tSubject2\tSubject3\tTotal\tPercentage\n");

    for (i = 0; i < n; i++)
    {
        printf("%d\t%s\t%.0f\t\t%.0f\t\t%.0f\t\t%.0f\t%.2f\n",
               s[i].roll_no,
               s[i].name,
               s[i].marks1,
               s[i].marks2,
               s[i].marks3,
               s[i].total,
               s[i].percentage);
    }

    return 0;
}
